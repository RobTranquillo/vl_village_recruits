# Number of emeralds needed to hire a villager (free in creative mode).
village_recruits_cost (Hiring cost in emeralds) int 3 0 64

# Maximum number of recruits a single player may own.
village_recruits_max_per_player (Max recruits per player) int 10 1 100

# Following recruits further away than this are teleported to their owner.
village_recruits_teleport_distance (Teleport distance) int 16 6 64

# Runs a self test after server start and logs the result (for development).
village_recruits_selftest (Run self test on startup) bool false

# Converts recruits from the pre-release development version (mod name
# "vl_recruits") when a world is loaded. Only needed for such old worlds.
village_recruits_convert_vl_recruits (Convert recruits from vl_recruits) bool false
